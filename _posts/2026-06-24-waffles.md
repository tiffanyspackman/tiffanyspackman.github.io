recipe:
  servings: 6-8 Servings
  prep: 30 minutes
  ingredients_markdown: |-

    * 2 cups flour
    * 1 Tbs. sugar
    * 1 tsp. salt
    * 3 tsp. baking powder
    * 2 egg whites beaten stiff
    * 2 cups milk
    * 4 Tbs. melted butter

  directions_markdown: |-
    
    1. Mix dry ingredients
    2. Add egg yolks and milk; beat
    3. Add butter
    4. Last, fold in the beaten egg whites
